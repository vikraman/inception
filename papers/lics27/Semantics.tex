The standard denotational semantics of the fine-grained calculus~\cite{Levy}
relies on categorical structure provided by a cartesian closed category
equipped with a strong monad~\cite{Moggi}.  While the cartesian closed
structure gives interpretation to type structure, the strong monad
encapsulates and provides interpretation for effectful computations.  The
theory of algebraic effects~\cite{PlotkinPower} further advocates the
consideration of computational monads that arise from algebraic theories.  In
this context, the operators of a theory turn into computational constructs
whose behavioural laws are specified by the equational presentation of the
theory.  The inception calculus was designed from first principles from this
viewpoint from the theory of inception algebras~\cite{FioreIHP2014}

\begin{definition}
For exponentiable objects $L$ and $P$ in a cartesian category, an
$(L,P)$-inception algebra is a structure
\[
\mathrm{rec} : L\times P \to A \leftarrow A^L\times A^P : \mathrm{inc}
\]
satisfying\\
%
$\begin{array}{rll}
\mathrm{(substitution)}
&
p:P , h:A^P
\vdash
\mathrm{inc}
 \big(\,
   \lambda\,{\ell\!:\!L}.\ \mathrm{rec}(\ell,p) 
   \,,\, 
   h 
 \,\big)
 =
 h(p)
& \quad
\begin{array}{c}\xymatrix{
  P\times A^P \ar[d] \ar[dr] \\
  A^L \times A^P \ar[r] & A
}\end{array}
\end{array}$
%
\\
%
$\begin{array}{rll}
\mathrm{(extensionality)}
& f: A^L, \ell: L\vdash 
\mathrm{inc}\big(\, f \,,\, \lambda\,{p\!:\!P}.\ \mathrm{rec}(\ell,p) \,\big) 
= 
f(\ell)
& \enspace
\begin{array}{c}\xymatrix{
  A^L\times L \ar[d] \ar[dr] \\
  A^L \times A^P \ar[r] & A
}\end{array}
\end{array}$
%
\\
%
$\begin{array}{rll} 
\mathrm{(weakening)}
& a:A, h: A^P \vdash \mathrm{inc}(\, \lambda\,{\ell\!:\!L}.\ a \,,\, h \,) =  a
& \qquad
\begin{array}{c}\xymatrix{
  A\times A^P \ar[d] \ar[dr] \\
  A^L \times A^P \ar[r] & A
}\end{array}
\end{array}$
%
\\
%
$\begin{array}{l} 
\mathrm{(associativity)}
\\[1mm]
f: (A^L\times A^P)^L , h: A^P
\\[1mm]
\ \vdash 
\mathrm{inc}
  \big(\, 
    \lambda\,{\ell\!:\!L}.\,\mathrm{inc} (f\,\ell) 
    \,,\, 
    h
  \,\big)
  = 
  \mathrm{inc}\big(\,
    \lambda\,{k\!:\!L}.\,
      \mathrm{inc}\big(\,
        \lambda\,{\ell\!:\!L}.\,\pi_1(f\ell)k
        \,,\,
        h
      \,\big)
    \,,\,
      \lambda\,{p\!:\!P}.\,
        \mathrm{inc}\big(\, \pi_2(f\ell)p ,\, h \,\big)
  \,\big)
\end{array}$
%
\[\xymatrix{
  (A^L \times A^P)^L \times A^P \ar[d] \ar[r]
  & (A^L\times A^P)^L \times (A^L\times A^P)^P
  \ar[r] & A^L\times A^P \ar[d]
  \\
  A^L \times A^P \ar[rr] & & A
}\]
\end{definition}

\begin{proposition}
A $(V,1)$-inception algebra in $\Set^\F$ is equivalently a substitution

\end{proposition}
