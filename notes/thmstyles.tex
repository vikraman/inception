\declaretheoremstyle[
  headpunct={},
  headfont=\normalfont\bfseries,
  numbered=yes,
  bodyfont=\normalfont,
  spaceabove=1em plus 0.75em minus 0.75em,
  spacebelow=1em plus 0.75em minus 0.75em,
  qed=\rule{0.6ex}{0.6ex}
]{thstyle}
